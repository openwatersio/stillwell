import { rows, json } from "../../../lib/store";
import { USER_AGENT } from "../../../lib/paths";

export const sources = await json("sources.json");
export const [totals] = await rows(`
  SELECT count(*)::INT AS stations, count(*) FILTER (WHERE status = 'active')::INT AS active,
    count(*) FILTER (WHERE live)::INT AS live
  FROM 'https://stillwell.openwaters.io/stations.parquet'`);
// A source with no levels written yet has no years file, so a 404 leaves it out. Any other failure stops the
// build, rather than publishing a total short of a source.
export const yearFiles = (
  await Promise.all(
    sources.map(async (s) => {
      const file = `https://stillwell.openwaters.io/stations/years/${s.id}.parquet`;
      const response = await fetch(file, { method: "HEAD", headers: { "User-Agent": USER_AGENT } });
      if (response.status === 404) return undefined;
      if (!response.ok) throw new Error(`${file}: HTTP ${response.status}`);
      return `'${file}'`;
    }),
  )
).filter(Boolean);
export const [observed] = await rows(`
  SELECT sum(good + interpolated + preliminary + suspect + bad)::DOUBLE AS values, min(year) AS first
  FROM read_parquet([${yearFiles.join(", ")}])`);
export const n = (x) => x.toLocaleString("en-US");
export const billions = (x) =>
  `${(x / 1e9).toLocaleString("en-US", { maximumFractionDigits: 1 })} billion`;

Stillwell is an open store of observed and predicted water levels from tide, river and lake gauges around the world. It contains [{n(totals.stations)} stations](/docs/guides/stations/) from [{sources.length} sources](/docs/sources/), with [{billions(observed.values)} observations](/docs/reference/observations/) going back to {observed.first}, published in one layout, with [quality control](/docs/guides/quality-control/), offsets to other [datums](/docs/guides/datums/).

It is built by [Open Waters](https://openwaters.io) to power tide predictions and water level observations for [Slackwater](https://slackwater.xyz), and to reduce crowd-sourced depth soundings to chart datum.

## What's in it?

- [**Sources**](/docs/sources/): the agencies and datasets the data comes from, with what each one covers and the terms its data comes under.
- [**Stations**](/docs/reference/stations/): every station's name, location, the kind of water it measures, its datum, how far back its record goes, whether it's still reporting, and the terms its data comes under.
- [**Measured water levels**](/docs/reference/observations/#observed-levels): what each gauge recorded, every minute to every hour depending on the source, as far back as the source's record goes.
- [**Predicted water levels**](/docs/reference/predictions/): the source's own predicted water level and high and low tides, for sources that publish them.
- [**Datums and offsets**](/docs/guides/datums/): what to add to a station's levels to refer them to another reference, such as mean sea level, chart datum or a national height system. Where two sources carry the same gauge in different datums, Stillwell [measures the offset between them](/docs/reference/links/) to fill in datums a station lacks.
- [**Known errors**](/docs/reference/errata/) found in what sources publish, such as a clock an hour off or a datum that shifted without notice, and how each is resolved.

Not every station has every kind of data. Its entry in the station list says which it has. History is still being filled in for several sources, so some stations have no values yet.
